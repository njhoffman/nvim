*_lua-metatables.txt* Template for making local helpfiles


                                  >lua
  setmetatable(tbl, {
    __index = {
      len = function(len)
        local incr = 0
        for _ in pairs(len) do
          incr = incr + 1
        end
        return incr
      end,
    },
  })
  print(tbl:len())
<
